export function useOptimisticUpdate<T>(
  initialData: T,
  updateFn: (data: T) => T
) {
  const [optimistic, setOptimistic] =
    useState(initialData);

  return { optimistic, update: (d: T) => {
    setOptimistic(updateFn(d));
  }};
}
